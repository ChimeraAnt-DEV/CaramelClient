# Baseline profile for the startup path.
#
# This is a hand-authored profile, not a Macrobenchmark capture: the device that would produce a
# capture is not available in this repository. It names the classes and methods the cold start
# touches, so ProfileInstaller compiles them ahead of time on device and the first frame is not
# paying for JIT. It is deliberately conservative - a profile that names a method that no longer
# exists is ignored rather than fatal, so it degrades quietly as the code moves.
#
# Regenerate a real profile with a Macrobenchmark module once one is added, and keep this file as
# the fallback for builds without it.

# Application and startup.
HSPLorg/chimeramc/client/core/minecraft/LauncherApplication;-><init>()V
HSPLorg/chimeramc/client/core/minecraft/LauncherApplication;->onCreate()V

# The splash screen, which is the first thing the user sees.
HSPLorg/chimeramc/client/ui/activities/SplashActivity;-><init>()V
HSPLorg/chimeramc/client/ui/activities/SplashActivity;->onCreate(Landroid/os/Bundle;)V
HSPLorg/chimeramc/client/launcher/ui/splash/SplashInit;->run(Landroid/content/Context;Lorg/chimeramc/client/launcher/ui/splash/SplashInit$Listener;)V

# The main screen the splash routes to.
HSPLorg/chimeramc/client/ui/activities/MainActivity;-><init>()V
HSPLorg/chimeramc/client/ui/activities/MainActivity;->onCreate(Landroid/os/Bundle;)V

# The shared activity shell every screen inflates through.
HSPLorg/chimeramc/client/ui/activities/BaseActivity;-><init>()V
HSPLorg/chimeramc/client/ui/activities/BaseActivity;->onCreate(Landroid/os/Bundle;)V

# Personalization is read on the first frame of every screen.
HSPLorg/chimeramc/client/launcher/util/PersonalizationManager;->isDarkMode(Landroid/content/Context;)Z

# The design tokens and the shared type/shape/motion entry points.
HSPLorg/chimeramc/client/ui/components/UiTokens;-><clinit>()V
